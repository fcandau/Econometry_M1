% ====================================================================
%  DiD deck — modern restyle (Metropolis-inspired, pdflatex-friendly)
%    - clean sans typography (Fira Sans + matching sans math)
%    - soft, low-saturation palette; flat tinted blocks (no full-red boxes)
%    - slim section-dot navigation at the top ("where are we")
% ====================================================================

% ---------- math packages ----------
\usepackage{amsmath}
\usepackage{amssymb}

% ---------- typography: Palatino text + matching Palatino math (mathpazo) ----------
\usepackage{mathpazo}                % Palatino roman + Pazo math
\usefonttheme{serif}                 % use the roman (Palatino) family for slide text

\usepackage{booktabs}
\usepackage{array}
\usepackage{bm}
\usepackage{tikz}
\usetikzlibrary{arrows.meta}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

% ---------- soft, low-saturation palette ----------
\definecolor{bordeaux}{rgb}{0.68,0.0,0.13}   % kept for compatibility
\definecolor{accent}{RGB}{151,30,47}         % softened bordeaux (brand)
\definecolor{accentsoft}{RGB}{197,150,158}   % muted accent (inactive nav)
\definecolor{ink}{RGB}{38,38,43}             % near-black body text
\definecolor{paneltint}{RGB}{245,240,241}    % very light warm grey (block bg)
\definecolor{alerttint}{RGB}{250,235,236}    % light red tint (alerts)
\definecolor{exatint}{RGB}{236,243,239}      % light green-grey tint (examples)

\setbeamercolor{normal text}{fg=ink,bg=white}
\setbeamercolor{structure}{fg=accent}
\setbeamercolor{alerted text}{fg=accent}

% ---------- top navigation: slim section dots ("where are we") ----------
\useoutertheme[subsection=false]{miniframes}
\setbeamercolor{section in head/foot}{fg=accent,bg=white}
\setbeamercolor{mini frame}{fg=accent,bg=accentsoft!45}
\setbeamerfont{section in head/foot}{size=\tiny}
% slim the nav bar: drop the separation lines and extra padding (reclaims height)
\setbeamertemplate{headline}{%
  \begin{beamercolorbox}{section in head/foot}\vskip1pt\insertnavigation{\paperwidth}\vskip1pt\end{beamercolorbox}%
}

% ---------- flat, modern frametitle: title + thin rule (no full bar) ----------
\setbeamercolor{frametitle}{fg=accent,bg=white}
\setbeamerfont{frametitle}{series=\bfseries}
\setbeamertemplate{frametitle}{%
  \vskip0.25em
  {\usebeamerfont{frametitle}\usebeamercolor[fg]{frametitle}\insertframetitle\par}%
  \vskip0.05em{\color{accent}\rule{\textwidth}{0.8pt}}\vskip0.06em
}

% ---------- soft blocks (tinted, no heavy fill) ----------
\setbeamertemplate{blocks}[rounded][shadow=false]
\setbeamercolor{block title}{fg=accent,bg=paneltint}
\setbeamercolor{block body}{fg=ink,bg=paneltint!55}
\setbeamercolor{block title alerted}{fg=accent,bg=alerttint}
\setbeamercolor{block body alerted}{fg=ink,bg=alerttint!60}
\setbeamercolor{block title example}{fg=accent!75!black,bg=exatint}
\setbeamercolor{block body example}{fg=ink,bg=exatint!60}

% ---------- list bullets in accent ----------
\setbeamertemplate{itemize items}[circle]
\setbeamercolor{itemize item}{fg=accent}
\setbeamercolor{itemize subitem}{fg=accentsoft}
\setbeamercolor{enumerate item}{fg=accent}

% ---------- minimal footline: frame number only ----------
\setbeamertemplate{navigation symbols}{}
\setbeamertemplate{footline}{%
  \hfill\usebeamercolor[fg]{section in head/foot}%
  {\scriptsize\insertframenumber\,/\,\inserttotalframenumber\hspace{1.4em}}\vskip4pt%
}

% ---------- softer section-divider box (less aggressive than full fill) ----------
\setbeamercolor{section title}{fg=accent,bg=paneltint}
\setbeamercolor{section name}{fg=accentsoft}

% ---------- softer title page (accent text, no full-red box) ----------
\setbeamercolor{title}{fg=accent,bg=}
\setbeamercolor{subtitle}{fg=ink,bg=}
\setbeamercolor{author}{fg=ink,bg=}
\setbeamercolor{titlelike}{fg=accent,bg=}

% ---------- handy macros (shared across the course) ----------
\newcommand{\Cov}{\operatorname{Cov}}
\newcommand{\Var}{\operatorname{Var}}
\newcommand{\Corr}{\operatorname{Corr}}
\newcommand{\E}{\mathbb{E}}
\newcommand{\indep}{\perp\!\!\!\perp}

% ---------- tighten the bibliography: kill the blank line between entries ----------
% pandoc already defined CSLReferences above this point, so redefine it directly
% (no \AtBeginDocument — that would mangle the #1/#2 parameters) with itemsep = 0.
\renewenvironment{CSLReferences}[2]{%
  \begin{list}{}{%
    \setlength{\itemindent}{0pt}%
    \setlength{\leftmargin}{0pt}%
    \setlength{\topsep}{0pt}%
    \setlength{\partopsep}{0pt}%
    \setlength{\parskip}{0pt}%
    \setlength{\parsep}{0pt}%
    \setlength{\itemsep}{0pt}%
    \ifodd #1\relax
      \setlength{\leftmargin}{\cslhangindent}%
      \setlength{\itemindent}{-1\cslhangindent}%
    \fi}%
}{\end{list}}
